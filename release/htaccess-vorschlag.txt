# Vorschlag für /wunderhaus/media/.htaccess  (EMFD-MEDIA-CORE-01 · Cache-Strategie)
# NICHT ungeprüft hochladen. Falls auf dem Server bereits eine .htaccess in /wunderhaus/media/
# oder /wunderhaus/ liegt, diese zuerst sichern und die Regeln unten nur ergänzen.
#
# Befund 18.11.2026: Der Server (Apache, STRATO) sendet für manifest.json, Bilder und JSON
# KEIN Cache-Control, nur ETag + Last-Modified. Browser cachen dann heuristisch
# (ca. 10 % des Dateialters) und fragen in dieser Zeit nicht nach -> ein neues Manifest
# wäre je nach Alter erst nach Stunden bis Tagen überall sichtbar.

<IfModule mod_headers.c>
  # 1. Manifest und alle JSON-Register: immer beim Server nachfragen (ETag/Last-Modified -> 304)
  <FilesMatch "\.(json)$">
    Header set Cache-Control "no-cache"
  </FilesMatch>

  # 2. Loader und Katalog: ebenfalls revalidieren (Name ändert sich bei Updates nicht)
  <FilesMatch "^(media-loader\.js|media-loader\.mjs|index\.html)$">
    Header set Cache-Control "no-cache"
  </FilesMatch>

  # 3. Alte Dateinamen ohne Hash (cards/O01.png, rooms/*.webp, audio/O01.mp3 ...): revalidieren,
  #    weil sie bereits einmal an Ort und Stelle ersetzt wurden (cards/O01-O18.png am 17.11.2026).
  <FilesMatch "\.(png|jpe?g|webp|svg|mp3)$">
    Header set Cache-Control "no-cache"
  </FilesMatch>

  # 4. Versionierte Dateien mit Revision + Hash im Namen (…-r2-<hash>.webp, …-front-r2-<hash>.png,
  #    …-v1-<hash>.webp): Inhalt ändert sich nie -> ein Jahr, unveränderlich.
  #    Muss NACH Regel 3 stehen, damit sie für diese Dateien gewinnt.
  <FilesMatch "-(r[0-9]+|v[0-9]+)-[0-9a-f]{8,10}\.(png|jpe?g|webp)$">
    Header set Cache-Control "public, max-age=31536000, immutable"
  </FilesMatch>
</IfModule>

# MIME-Typen absichern (heute korrekt geliefert; schadet nicht)
<IfModule mod_mime.c>
  AddType image/webp .webp
  AddType audio/mpeg .mp3
  AddType application/json .json
  AddType text/javascript .mjs
</IfModule>

# Kontrolle nach dem Hochladen (lesend):
#   curl -sI https://flowisaurus.de/wunderhaus/media/manifest.json | grep -i cache-control      -> no-cache
#   curl -sI https://flowisaurus.de/wunderhaus/media/optimized/cards/O01-630x1080-r2-40012d6477.webp | grep -i cache-control   -> immutable
# Rollback: .htaccess wieder entfernen bzw. Sicherung zurückspielen.
